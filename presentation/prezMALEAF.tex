    \begin{frame}{Sujet du stage}

        \begin{itemize}
            \item \textbf{Pompes centrifuges} : équipements généralement industriels, utilisés dans divers secteurs (eau, pétrole, chimie, énergie).
            \item \textbf{mécanique des fluides numérique numérique} : important pour la prédiction du comportement physique, la conception et l'optimisation, mais confrontée à la complexité des écoulements.
            \item \textbf{Optimisation de la performance} : essentielle pour réduire la consommation énergétique et les coûts opérationnels.
            \item \textbf{Objectif du stage} : améliorer la performance d'une pompe centrifuge en utilisant des techniques d'assimilation de données (DA) pour ajuster les paramètres du modèle numérique.
        \end{itemize}

    \end{frame}


    \begin{frame}{Numérique et Expérimental}
        \begin{columns}[t] % alignement en haut
            % Colonne numérique
            \begin{column}{0.45\textwidth}
                \begin{block}{Simulation numérique}
                    \begin{itemize}
                        \item Couverture complète (champ de vitesses, pressions, etc.)
                        \item Peu coûteux une fois le modèle construit
                        \item Mais : précision limitée par modèles (turbulence, maillage, CL)
                    \end{itemize}
                \end{block}
            \end{column}

            % Colonne expérimental
            \begin{column}{0.45\textwidth}
                \begin{block}{Expérimental}
                    \begin{itemize}
                        \item Mesures réelles et précises
                        \item Validation indispensable des modèles
                        \item Mais : données locales, coût et mise en place difficiles
                    \end{itemize}
                \end{block}
            \end{column}
        \end{columns}

    \vspace{0.5cm}
    \centering
        \begin{block}{Assimilation de données}
            Combiner \textbf{numérique} (complet mais approximatif) et \textbf{expérimental} (précis mais limité) 
            pour obtenir des simulations plus fiables et robustes.
        \end{block}
    \end{frame}

    \begin{frame}{Modélisation avec la méthode de frontières immergées (IBM)}

        \begin{columns}
            \hspace{-0.5cm}
            \begin{column}{0.5\linewidth}
            \begin{itemize}[<+->]
                \item Simulation stationnaire avec l'approche \texorpdfstring{\structure{\textbf{Multiple Reference Frame (MRF)}}}{Multiple Reference Frame (MRF)}
                \item Modèle de turbulence $\mathcal{K}-\omega$ SST $\rightarrow$ Cinq coefficients de pénalisation $D = (\boldsymbol{D_u}, D_\mathcal{K}, D_\omega)$
                

            \begin{eqnarray*}
                \boldsymbol{f}_{P_u} &= \xi \nu \boldsymbol{D} (\boldsymbol{u}-\boldsymbol{u_{ib}}) \rightarrow \boldsymbol{u_{ib}} &= \cancel{\boldsymbol{u}_r} + \boldsymbol{\Omega} \times \boldsymbol{r} \\
                \boldsymbol{f}_{P_\mathcal{K}} &= \xi \nu D_{\eta} (\mathcal{K}-\mathcal{K}_{ib}) \rightarrow \mathcal{K}_{ib} &= 0 \\
                \boldsymbol{f}_{P_\omega} &= \xi \nu D_{\omega} (\omega-\omega_{ib}) \rightarrow \omega_{ib} &= (\omega_{vis}^2+\omega_{log}^2)^{1/2}
            \end{eqnarray*}
            où $\omega_{vis}=\frac{6 \nu}{\beta_1 y_{\mathrm{ref}}^2}$ et $\omega_{log} = \frac{\sqrt{\mathcal{K}}}{C_\mu \kappa y_\mathrm{ref}}$

            \vspace{0.5cm}

            \end{itemize}
